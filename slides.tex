% Slide 2: Key Highlights
\begin{frame}{Key Highlights of the Talks}
  \begin{itemize}
    \item \textbf{India--Canada Defence Dialogue} established to enhance defence industries, maritime domain awareness, and military exchanges
    \item \textbf{Trade Target:} Ambitious goal to expand bilateral trade to \textbf{\$50 billion by 2030}
    \item \textbf{CEPA:} Both sides agreed to expedite the Comprehensive Economic Partnership Agreement
    \item \textbf{Counter-Terrorism:} Close cooperation against terrorism, extremism, and radicalization identified as vital for global peace
    \item Both nations reaffirmed shared commitment to \textbf{democratic values} and diversity
  \end{itemize}
\end{frame}

% Slide 3: Economic & Strategic Outcomes
\begin{frame}{Economic \& Strategic Outcomes}
  \begin{itemize}
    \item \textbf{Three Agreements Signed:}
      \begin{itemize}
        \item Critical Minerals cooperation
        \item Renewable Energy partnership
        \item Cultural exchange
      \end{itemize}
    \item \textbf{Clean Energy \& Fintech} cooperation identified as defining pillars of the visit
    \item Leaders attended the \textbf{India--Canada CEOs Forum} to strengthen commercial ties
    \item Canada views \textbf{energy cooperation} as a cornerstone of the renewed relationship
    \item External Affairs Minister Jaishankar welcomed Canadian Foreign Minister \textbf{Anita Anand} in New Delhi
  \end{itemize}
\end{frame}

% Slide 4: A New Chapter in Relations
\begin{frame}{A New Chapter in India--Canada Relations}
  \begin{itemize}
    \item Canadian PM Carney's \textbf{first full-fledged bilateral visit} to India --- a major reset after strained ties
    \item Visit dates: \textbf{February 26 -- March 7, 2026}
    \item Built on earlier meetings:
      \begin{itemize}
        \item \textbf{G7 Summit} --- Kananaskis, Canada (June 2025)
        \item \textbf{G20 Summit} --- Johannesburg, South Africa (November 2025)
      \end{itemize}
    \item PM Carney: \textit{``Canada is renewing its relationship with India and expanding its ties, building a strong foundation in trade, defence, and energy''}
    \item The visit marks an \textbf{inflection point} in the India--Canada Strategic Partnership
  \end{itemize}
\end{frame}
