% latex-preamble.tex
% Shared LaTeX preamble included in all deckpipe PDF outputs.
% Passed to pandoc via -H (--include-in-header).

\usepackage{xcolor}
\usepackage{hyperref}
\usepackage{microtype}

% ── Palette ───────────────────────────────────────────────────────────────────
\definecolor{dpNavy}{HTML}{003A70}
\definecolor{dpMidBlue}{HTML}{0077B6}
\definecolor{dpTeal}{HTML}{00B4D8}

% ── Hyperlinks ────────────────────────────────────────────────────────────────
\hypersetup{
  colorlinks = true,
  linkcolor  = dpNavy,
  urlcolor   = dpMidBlue,
  citecolor  = dpTeal,
  pdfborder  = {0 0 0}
}

% ── Fonts (report PDF only; Beamer theme handles its own font setup) ──────────
% Only runs outside Beamer. Explicitly load fontspec so \IfFontExistsTF is
% available regardless of whether pandoc's template loaded it already.
\ifdefined\beamer@version\else
  \usepackage{fontspec}
  \IfFontExistsTF{Calibri}{%
    \setmainfont{Calibri}[
      BoldFont      = Calibri Bold,
      ItalicFont    = Calibri Italic,
      BoldItalicFont = Calibri Bold Italic
    ]
    \setsansfont{Calibri}[
      BoldFont      = Calibri Bold,
      ItalicFont    = Calibri Italic,
      BoldItalicFont = Calibri Bold Italic
    ]
  }{%
    % Calibri not found — use TeX Gyre Heros (Helvetica clone, ships with texlive)
    \setmainfont{TeX Gyre Heros}
    \setsansfont{TeX Gyre Heros}
  }
\fi
